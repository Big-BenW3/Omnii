//! PER Privacy Proof Tests - Phase 1 Gates C, D, E
//!
//! This test module provides the test program for verifying MagicBlock PER privacy guarantees.
//! These tests MUST pass on the deployed Devnet PER environment before claiming private voting works.
//!
//! Gates to prove:
//! - Gate C: Private Ballot Isolation - voters cannot read each other's ballots
//! - Gate D: Tally Completeness - program can read all ballots, voters cannot
//! - Gate E: No Premature Public Result - summary only visible after commit
//!
//! Run with: cargo test --test per_privacy_tests

use solana_program::pubkey::Pubkey;
use solana_sdk::signature::Keypair;
use std::str::FromStr;

// Test constants from spec
const MAX_ELIGIBLE_VOTERS: usize = 20;

/// MagicBlock PER Devnet configuration
/// These values MUST be obtained from official MagicBlock docs
pub struct MagicBlockPerConfig {
    pub er_rpc_url: String,
    pub per_rpc_url: String,
    pub queue_addresses: Vec<Pubkey>,
    pub validator_address: Option<Pubkey>,
}

impl Default for MagicBlockPerConfig {
    fn default() -> Self {
        // Default values from spec Section 12.3
        Self {
            er_rpc_url: "https://devnet.magicblock.app/api/er/rpc".to_string(),
            per_rpc_url: "https://devnet.magicblock.app/api/per/rpc".to_string(),
            queue_addresses: vec![
                // VRF queue addresses from spec
                Pubkey::from_str("Cuj97ggrhhidhbu39TijNVqE74xvKJ69gDervRUXAxGh").unwrap(),
                Pubkey::from_str("5hBR571xnXppuCPveTrctfTU7tJLSN94nq7kv7FRK5Tc").unwrap(),
            ],
            validator_address: None,
        }
    }
}

/// Test setup for PER environment
pub struct PerTestSetup {
    pub config: MagicBlockPerConfig,
    pub room_keypair: Keypair,
    pub host_keypair: Keypair,
    pub voter_keypairs: Vec<Keypair>,
}

impl PerTestSetup {
    pub fn new(num_voters: usize) -> Self {
        let mut voter_keypairs = Vec::with_capacity(num_voters);
        for _ in 0..num_voters {
            voter_keypairs.push(Keypair::new());
        }

        Self {
            config: MagicBlockPerConfig::default(),
            room_keypair: Keypair::new(),
            host_keypair: Keypair::new(),
            voter_keypairs,
        }
    }
}

/// Helper functions for PER testing
pub mod per_helpers {
    use super::*;

    /// Generate a test room setup with the specified number of voters
    pub fn generate_test_setup(num_voters: usize) -> PerTestSetup {
        assert!(num_voters <= MAX_ELIGIBLE_VOTERS, "Too many voters");
        PerTestSetup::new(num_voters)
    }

    /// Get MagicBlock PER configuration from environment or defaults
    pub fn get_per_config() -> MagicBlockPerConfig {
        MagicBlockPerConfig::default()
    }
}
